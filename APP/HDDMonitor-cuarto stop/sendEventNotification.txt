const functions = require("firebase-functions");
const admin = require("firebase-admin");
admin.initializeApp();

exports.sendEventNotification = functions.firestore
    .document('hdd-monitor/accounts/clients/{clientId}/events/{eventId}')
    .onWrite(async (change, context) => {
        console.log('Procesando cambio en evento');
        
        const clientId = context.params.clientId;
        const eventId = context.params.eventId;

        try {
            // Obtener información del cliente
            const clientDoc = await admin.firestore()
                .doc(`hdd-monitor/accounts/clients/${clientId}`)
                .get();

            if (!clientDoc.exists) {
                console.error('Cliente no encontrado:', clientId);
                return null;
            }

            const clientName = clientDoc.data().name;
            console.log('Información del cliente:', { clientName, clientId });

            // Si el documento fue eliminado
            if (!change.after.exists) {
                const deletedData = change.before.data();
                return handleEventDeletion(deletedData, clientId, clientName, eventId);
            }

            // Si es una creación o actualización
            const newValue = change.after.data();
            if (!change.before.exists) {
                return handleEventCreation(newValue, clientId, clientName, eventId);
            }

            // Si es una actualización
            const previousValue = change.before.data();
            // Verificar cambios significativos
            const changedFields = [];
            Object.entries(newValue).forEach(([key, value]) => {
                if (key !== 'lastUpdate' && JSON.stringify(value) !== JSON.stringify(previousValue[key])) {
                    changedFields.push(key);
                }
            });

            if (changedFields.length === 0) {
                console.log('No hay cambios significativos');
                return null;
            }

            console.log('Campos modificados:', changedFields);
            return handleEventUpdate(newValue, previousValue, clientId, clientName, eventId, changedFields);

        } catch (error) {
            console.error('Error general en procesamiento:', error);
            return null;
        }
    });

async function handleEventDeletion(deletedData, clientId, clientName, eventId) {
    try {
        const actorInfo = await getActorInfo(deletedData.createdByUserId, deletedData.createdByUserRole, clientId);
        
        const basePayload = {
            data: {
                clientDocName: clientId,
                eventId: eventId,
                type: 'event_deleted'
            },
            android: {
                priority: 'high',
                notification: {
                    channelId: 'event_notifications',
                    priority: 'high',
                    sound: 'default',
                    visibility: 'public'
                }
            }
        };

        await sendNotifications(
            clientId,
            {
                title: 'Evento Eliminado',
                body: actorInfo.role === 'ADMIN' ?
                    `Un administrador ha eliminado el evento "${deletedData.title}"` :
                    `${actorInfo.name} ha eliminado el evento "${deletedData.title}"`
            },
            {
                title: `${clientName} - Evento Eliminado`,
                body: actorInfo.role === 'ADMIN' ?
                    `${actorInfo.name} (Admin) ha eliminado el evento "${deletedData.title}" (Cliente: ${clientName})` :
                    `${actorInfo.name} (Usuario de ${clientName}) ha eliminado el evento "${deletedData.title}"`
            },
            basePayload,
            deletedData.createdByUserId
        );
    } catch (error) {
        console.error('Error en handleEventDeletion:', error);
    }
}

async function handleEventUpdate(newValue, previousValue, clientId, clientName, eventId, changedFields) {
    try {
        const actorInfo = await getActorInfo(newValue.createdByUserId, newValue.createdByUserRole, clientId);
        
        let title, userMessage, adminMessage;

        if (changedFields.includes('status')) {
            const messages = getStatusChangeMessages(newValue, previousValue, actorInfo, clientName);
            title = messages.title;
            userMessage = messages.userMessage;
            adminMessage = messages.adminMessage;
        } else {
            title = `${clientName} - Evento Modificado`;
            userMessage = actorInfo.role === 'ADMIN' ?
                `Un administrador ha modificado el evento "${newValue.title}"` :
                `${actorInfo.name} ha modificado el evento "${newValue.title}"`;
            adminMessage = actorInfo.role === 'ADMIN' ?
                `${actorInfo.name} (Admin) ha modificado el evento "${newValue.title}" (Cliente: ${clientName})` :
                `${actorInfo.name} (Usuario de ${clientName}) ha modificado el evento "${newValue.title}"`;
        }

        const basePayload = {
            data: {
                clientDocName: clientId,
                eventId: eventId,
                type: 'event_updated',
                newStatus: newValue.status,
                oldStatus: previousValue.status,
                changedFields: changedFields.join(',')
            },
            android: {
                priority: 'high',
                notification: {
                    channelId: 'event_notifications',
                    priority: 'high',
                    sound: 'default',
                    visibility: 'public'
                }
            }
        };

        await sendNotifications(
            clientId,
            {
                title: title.replace(`${clientName} - `, ''),
                body: userMessage
            },
            {
                title: title,
                body: adminMessage
            },
            basePayload,
            newValue.createdByUserId
        );
    } catch (error) {
        console.error('Error en handleEventUpdate:', error);
    }
}

function getStatusChangeMessages(newValue, previousValue, actorInfo, clientName) {
    let title, userMessage, adminMessage;

    switch(newValue.status) {
        case 'ACEPTADO':
            title = `${clientName} - Evento Aceptado`;
            userMessage = actorInfo.role === 'ADMIN' ?
                `Un administrador ha aceptado el evento "${newValue.title}"` :
                `${actorInfo.name} ha aceptado el evento "${newValue.title}"`;
            adminMessage = actorInfo.role === 'ADMIN' ?
                `${actorInfo.name} (Admin) ha aceptado el evento "${newValue.title}" (Cliente: ${clientName})` :
                `${actorInfo.name} (Usuario de ${clientName}) ha aceptado el evento "${newValue.title}"`;
            break;

        case 'FINALIZADO':
            title = `${clientName} - Evento Finalizado`;
            userMessage = actorInfo.role === 'ADMIN' ?
                `Un administrador ha finalizado el evento "${newValue.title}"` :
                `${actorInfo.name} ha finalizado el evento "${newValue.title}"`;
            adminMessage = actorInfo.role === 'ADMIN' ?
                `${actorInfo.name} (Admin) ha finalizado el evento "${newValue.title}" (Cliente: ${clientName})` :
                `${actorInfo.name} (Usuario de ${clientName}) ha finalizado el evento "${newValue.title}"`;
            break;

        case 'PROGRAMADO':
            if (previousValue.status === 'FINALIZADO') {
                title = `${clientName} - Evento Reabierto`;
                userMessage = actorInfo.role === 'ADMIN' ?
                    `Un administrador ha reabierto el evento "${newValue.title}"` :
                    `${actorInfo.name} ha reabierto el evento "${newValue.title}"`;
                adminMessage = actorInfo.role === 'ADMIN' ?
                    `${actorInfo.name} (Admin) ha reabierto el evento "${newValue.title}" (Cliente: ${clientName})` :
                    `${actorInfo.name} (Usuario de ${clientName}) ha reabierto el evento "${newValue.title}"`;
            }
            break;

        default:
            title = `${clientName} - Cambio de Estado`;
            userMessage = actorInfo.role === 'ADMIN' ?
                `Un administrador ha cambiado el estado del evento "${newValue.title}" de ${previousValue.status} a ${newValue.status}` :
                `${actorInfo.name} ha cambiado el estado del evento "${newValue.title}" de ${previousValue.status} a ${newValue.status}`;
            adminMessage = actorInfo.role === 'ADMIN' ?
                `${actorInfo.name} (Admin) ha cambiado el estado del evento "${newValue.title}" de ${previousValue.status} a ${newValue.status} (Cliente: ${clientName})` :
                `${actorInfo.name} (Usuario de ${clientName}) ha cambiado el estado del evento "${newValue.title}" de ${previousValue.status} a ${newValue.status}`;
    }

    return { title, userMessage, adminMessage };
}

async function getActorInfo(actorId, actorRole, clientId) {
    if (!actorId) return { name: 'Usuario desconocido', role: 'UNKNOWN' };

    try {
        const isAdmin = actorRole === 'ADMIN';
        const path = isAdmin 
            ? `hdd-monitor/accounts/admins/${actorId}`
            : `hdd-monitor/accounts/clients/${clientId}/users/${actorId}`;
        
        const actorDoc = await admin.firestore().doc(path).get();
        return {
            name: actorDoc.exists ? actorDoc.data().name : 'Usuario desconocido',
            role: actorRole
        };
    } catch (error) {
        console.error('Error obteniendo información del actor:', error);
        return { name: 'Usuario desconocido', role: 'UNKNOWN' };
    }
}

async function sendNotifications(clientId, userNotification, adminNotification, basePayload, excludeUserId) {
    try {
        // Enviar a usuarios del cliente
        const userSnapshot = await admin.firestore()
            .collection(`hdd-monitor/accounts/clients/${clientId}/users`)
            .get();

        for (const userDoc of userSnapshot.docs) {
            if (userDoc.data().fcmToken && userDoc.id !== excludeUserId) {
                try {
                    const userPayload = {
                        ...basePayload,
                        token: userDoc.data().fcmToken,
                        notification: userNotification
                    };
                    await admin.messaging().send(userPayload);
                    console.log('Notificación enviada a usuario:', userDoc.id);
                } catch (error) {
                    console.error('Error enviando notificación a usuario:', userDoc.id, error);
                }
            }
        }

        // Enviar a administradores
        const adminSnapshot = await admin.firestore()
            .collection('hdd-monitor/accounts/admins')
            .get();

        for (const adminDoc of adminSnapshot.docs) {
            if (adminDoc.data().fcmToken && adminDoc.id !== excludeUserId) {
                try {
                    const adminPayload = {
                        ...basePayload,
                        token: adminDoc.data().fcmToken,
                        notification: adminNotification
                    };
                    await admin.messaging().send(adminPayload);
                    console.log('Notificación enviada a admin:', adminDoc.id);
                } catch (error) {
                    console.error('Error enviando notificación a admin:', adminDoc.id, error);
                }
            }
        }
    } catch (error) {
        console.error('Error en sendNotifications:', error);
    }
}
